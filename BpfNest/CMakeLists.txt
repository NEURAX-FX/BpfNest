cmake_minimum_required(VERSION 3.22)
project(BpfNest VERSION 0.1.0 LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(BPFNEST_ZYGISK_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/third_party/zygisk_next_api.h")
if(NOT EXISTS "${BPFNEST_ZYGISK_HEADER}")
    message(FATAL_ERROR
        "Missing third_party/zygisk_next_api.h. The upstream ZygiskNext API4 header "
        "is not redistributed with this repository. Run scripts/fetch-api-header.sh "
        "from the BpfNest source directory, or place the pinned header there manually.")
endif()

add_library(bpfnest_core STATIC src/core.cpp src/install.cpp src/elf.cpp)
set_target_properties(bpfnest_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bpfnest_core PUBLIC include third_party)
target_compile_options(bpfnest_core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)

if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "BpfNest supports only arm64-v8a")
    endif()
    add_library(bpfnest SHARED src/runtime.cpp src/trampoline.S)
    target_include_directories(bpfnest PRIVATE third_party)
    target_link_libraries(bpfnest PRIVATE bpfnest_core log dl)
    target_compile_options(bpfnest PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror;-fno-exceptions;-fno-rtti;-fvisibility=hidden>)
    target_link_options(bpfnest PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack -Wl,-z,max-page-size=16384 -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map)
else()
    enable_testing()
    add_executable(bpfnest_core_test tests/core_test.cpp)
    target_link_libraries(bpfnest_core_test PRIVATE bpfnest_core)
    target_compile_options(bpfnest_core_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME core COMMAND bpfnest_core_test)
    add_executable(bpfnest_install_test tests/install_test.cpp)
    target_link_libraries(bpfnest_install_test PRIVATE bpfnest_core)
    target_compile_options(bpfnest_install_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME install COMMAND bpfnest_install_test)
    set(BPFNEST_ELF_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/../libnetd_updatable.so"
        CACHE FILEPATH "Optional real libnetd_updatable.so for the ELF certification test")
    add_executable(bpfnest_elf_test tests/elf_tests.cpp)
    target_link_libraries(bpfnest_elf_test PRIVATE bpfnest_core)
    target_compile_options(bpfnest_elf_test PRIVATE -Wall -Wextra -Werror)
    if(EXISTS "${BPFNEST_ELF_FIXTURE}")
        add_test(NAME elf COMMAND bpfnest_elf_test "${BPFNEST_ELF_FIXTURE}")
    else()
        message(STATUS
            "Skipping elf CTest: ${BPFNEST_ELF_FIXTURE} not found. "
            "Device firmware images are not redistributed; supply your own "
            "libnetd_updatable.so at the path above or set -DBPFNEST_ELF_FIXTURE=<path>.")
    endif()
    add_executable(bpfnest_fd_test tests/fd_test.cpp)
    target_link_libraries(bpfnest_fd_test PRIVATE bpfnest_core)
    target_compile_options(bpfnest_fd_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME fd COMMAND bpfnest_fd_test)
    add_executable(bpfnest_host_smoke tests/host_smoke.cpp)
    target_link_libraries(bpfnest_host_smoke PRIVATE bpfnest_core)
    target_compile_options(bpfnest_host_smoke PRIVATE -Wall -Wextra -Werror)
    add_test(NAME host_smoke COMMAND bpfnest_host_smoke)
    find_package(Threads REQUIRED)
    add_executable(bpfnest_reentrancy_test tests/reentrancy_test.cpp)
    target_link_libraries(bpfnest_reentrancy_test PRIVATE bpfnest_core Threads::Threads)
    target_compile_options(bpfnest_reentrancy_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME reentrancy COMMAND bpfnest_reentrancy_test)
    add_executable(bpfnest_policy_safety_test tests/policy_safety_test.cpp)
    target_link_libraries(bpfnest_policy_safety_test PRIVATE bpfnest_core)
    target_compile_options(bpfnest_policy_safety_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME policy_safety COMMAND bpfnest_policy_safety_test)
    add_executable(bpfnest_install_concurrency_test tests/install_concurrency_test.cpp)
    target_link_libraries(bpfnest_install_concurrency_test PRIVATE bpfnest_core Threads::Threads)
    target_compile_options(bpfnest_install_concurrency_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME install_concurrency COMMAND bpfnest_install_concurrency_test)
endif()
